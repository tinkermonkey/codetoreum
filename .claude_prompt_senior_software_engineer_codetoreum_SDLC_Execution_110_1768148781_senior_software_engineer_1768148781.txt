**Event Bus Architecture**:

The existing EventBus infrastructure (`src/codetoreum/infrastructure/event_bus.py`) routes events from adapters to handlers. Extend it to support adapter events:

```python
# src/codetoreum/infrastructure/event_bus.py (extend existing)
class EventBus:
    """Central event bus routing events to handlers."""
    
    def __init__(self):
        self._handlers: Dict[str, List[Callable]] = {}
        self._redis_client = redis.Redis()  # Existing infrastructure
    
    def subscribe(self, event_type: str, handler: Callable[[CodetoreumEvent], Awaitable[None]]) -> None:
        """Subscribe handler to event type."""
        if event_type not in self._handlers:
            self._handlers[event_type] = []
        self._handlers[event_type].append(handler)
    
    async def publish(self, event: CodetoreumEvent) -> None:
        """Publish event to all subscribers and persist."""
        # Persist to Redis Streams
        await self._redis_client.xadd(
            f"events:{event.type}",
            {"payload": json.dumps(asdict(event))}
        )
        
        # Route to handlers
        if event.type in self._handlers:
            for handler in self._handlers[event.type]:
                await handler(event)
        
        # Persist to Elasticsearch (existing infrastructure)
        await self._persist_to_elasticsearch(event)
```

**Workflow Engine Event Handlers** (update `src/codetoreum/application/workflow_orchestrator.py`):

```python
from codetoreum.domain.events.board_events import WorkItemColumnChangedEvent, BoardReconciledEvent
from codetoreum.domain.events.discussion_events import CommentNeedsResponseEvent
from codetoreum.domain.events.lock_events import LockReleasedEvent
from codetoreum.domain.events.review_events import ReviewStatusChangedEvent

class WorkflowOrchestrator:
    """Orchestrates workflow execution based on events."""
    
    def __init__(
        self,
        event_bus: EventBus,
        board_service: IBoardService,
        discussion_adapter: IDiscussionAdapter,
        lock_service: IPipelineLockService,
        review_service: ICodeReviewService,
        agent_executor: AgentExecutor,
        config_store: ConfigStore
    ):
        self._event_bus = event_bus
        self._board_service = board_service
        self._discussion_adapter = discussion_adapter
        self._lock_service = lock_service
        self._review_service = review_service
        self._agent_executor = agent_executor
        self._config_store = config_store
        
        # Subscribe to events
        self._subscribe_to_events()
    
    def _subscribe_to_events(self) -> None:
        """Subscribe workflow engine to adapter events."""
        self._event_bus.subscribe('workitem.column_changed', self._handle_column_change)
        self._event_bus.subscribe('comment.needs_response', self._handle_comment)
        self._event_bus.subscribe('lock.released', self._handle_lock_released)
        self._event_bus.subscribe('review.status_changed', self._handle_review_status)
    
    async def _handle_column_change(self, event: WorkItemColumnChangedEvent) -> None:
        """Handle work item column change event."""
        # Get column configuration
        column_config = await self._config_store.get_column_config(
            event.project_id, 
            event.board_id, 
            event.to_column
        )
        
        # Case 1: Automated column - trigger agent
        if column_config.type == 'automated' and column_config.agent:
            # Try to acquire pipeline lock
            success, reason = await self._lock_service.try_acquire_lock(
                event.project_id, 
                event.board_id, 
                event.work_item_id
            )
            
            if success:
                await self._agent_executor.trigger_agent(
                    work_item_id=event.work_item_id,
                    agent_name=column_config.agent,
                    context={
                        'column': event.to_column,
                        'project_id': event.project_id,
                        'board_id': event.board_id
                    }
                )
            else:
                # Queued for later execution (lock service handles queue)
                pass
        
        # Case 2: Exit column - release lock
        if column_config.exit_column:
            await self._lock_service.release_lock(
                event.project_id, 
                event.board_id, 
                event.work_item_id
            )
        
        # Case 3: Conversational column - start discussion monitoring
        if column_config.type == 'conversational' and column_config.agent:
            self._discussion_adapter.start_monitoring(
                event.work_item_id,
                DiscussionMonitoringConfig(
                    project_id=event.project_id,
                    column_name=event.to_column,
                    agent_assignment=column_config.agent
                )
            )
        
        # Case 4: Leaving conversational column - stop monitoring
        from_column_config = await self._config_store.get_column_config(
            event.project_id, 
            event.board_id, 
            event.from_column
        )
        if from_column_config.type == 'conversational':
            self._discussion_adapter.stop_monitoring(event.work_item_id)
    
    async def _handle_comment(self, event: CommentNeedsResponseEvent) -> None:
        """Handle comment requiring response."""
        # Trigger conversational agent
        await self._agent_executor.trigger_agent(
            work_item_id=event.work_item_id,
            agent_name=event.context.agent_assignment,
            context={
                'comment': event.comment,
                'context': event.context,
                'project_id': event.project_id
            }
        )
    
    async def _handle_lock_released(self, event: LockReleasedEvent) -> None:
        """Handle pipeline lock released event."""
        if event.next_in_queue:
            # Acquire lock for next queued item
            success, reason = await self._lock_service.try_acquire_lock(
                event.project_id,
                event.board_id,
                event.next_in_queue
            )
            
            if success:
                # Get column configuration and trigger agent
                position = await self._board_service.get_item_position(event.next_in_queue)
                column_name = position[0]
                column_config = await self._config_store.get_column_config(
                    event.project_id,
                    event.board_id,
                    column_name
                )
                
                if column_config.type == 'automated' and column_config.agent:
                    await self._agent_executor.trigger_agent(
                        work_item_id=event.next_in_queue,
                        agent_name=column_config.agent,
                        context={
                            'column': column_name,
                            'project_id': event.project_id,
                            'board_id': event.board_id
                        }
                    )
    
    async def _handle_review_status(self, event: ReviewStatusChangedEvent) -> None:
        """Handle code review status change."""
        if event.new_status == 'approved':
            # Move work item to next column in workflow
            if event.work_item_id:
                workflow = await self._config_store.get_workflow(event.project_id)
                next_column = workflow.get_next_column_after_review(event.work_item_id)
                
                if next_column:
                    await self._board_service.move_item_to_column(
                        event.work_item_id,
                        next_column
                    )
        
        elif event.new_status == 'changes_requested':
            # Move work item back to development column
            if event.work_item_id:
                workflow = await self._config_store.get_workflow(event.project_id)
                dev_column = workflow.get_development_column()
                
                await self._board_service.move_item_to_column(
                    event.work_item_id,
                    dev_column
                )
```

**Wiring Adapters to Event Bus** (create `src/codetoreum/infrastructure/event_bus_wiring.py`):

```python
def wire_adapters_to_event_bus(
    event_bus: EventBus,
    board_service: IBoardService,
    discussion_adapter: IDiscussionAdapter,
    lock_service: IPipelineLockService,
    review_service: ICodeReviewService
) -> None:
    """Wire adapter event emitters to central event bus."""
    
    # Board service events
    board_service.on('workitem.column_changed', lambda e: asyncio.create_task(event_bus.publish(e)))
    board_service.on('board.reconciled', lambda e: asyncio.create_task(event_bus.publish(e)))
    
    # Discussion adapter events
    discussion_adapter.on('comment.needs_response', lambda e: asyncio.create_task(event_bus.publish(e)))
    discussion_adapter.on('comment.posted', lambda e: asyncio.create_task(event_bus.publish(e)))
    
    # Lock service events
    lock_service.on('lock.acquired', lambda e: asyncio.create_task(event_bus.publish(e)))
    lock_service.on('lock.released', lambda e: asyncio.create_task(event_bus.publish(e)))
    lock_service.on('lock.stale_detected', lambda e: asyncio.create_task(event_bus.publish(e)))
    
    # Review service events
    review_service.on('review.status_changed', lambda e: asyncio.create_task(event_bus.publish(e)))
    review_service.on('review.comment_added', lambda e: asyncio.create_task(event_bus.publish(e)))
```

## Executive Summary

This architectural analysis addresses the implementation of vendor-agnostic abstraction layers for Codetoreum, enabling the orchestrator to work seamlessly with multiple external systems (GitHub, JIRA, Trello, Azure DevOps, etc.) without coupling to vendor-specific implementations. The core design principle—event-based adapters where detection mechanisms are adapter-internal and the orchestrator receives standardized events—aligns well with the existing hexagonal architecture and complements the established port/adapter pattern already implemented in the codebase.

## System Architecture

### Current State Assessment

The existing Codetoreum codebase already implements a hexagonal architecture with:
- **Output Ports**: `ITicketSystem`, `ILLMProvider`, `IContainer`, `IRepository`, `IEventStore` (all in `src/codetoreum/ports/output/`)
- **Adapters**: `GitHubTicketAdapter`, `ClaudeCodeAdapter`, `InMemoryTicketAdapter` (in `src/codetoreum/adapters/`)
- **Resilience Layer**: Circuit breakers, rate limiters, retry policies (in `src/codetoreum/infrastructure/resilience/`)
- **Event System**: Event bus, event store, event handlers (in `src/codetoreum/infrastructure/`)

## Scalability Design

### Event Distribution Architecture

The event system must support high-throughput event processing with multiple subscribers:

```python
# Event distribution via existing Redis infrastructure
┌────────────────────────────────────────────────────────────┐
│                  Event Flow Architecture                    │
│                                                             │
│  Adapters ──► Redis Pub/Sub ──► Event Bus ──► Handlers    │
│                    │                                        │
│                    └──► Redis Streams (buffering)          │
│                              │                              │
│                              └──► Elasticsearch (persist)  │
└────────────────────────────────────────────────────────────┘
```

**Throughput Considerations**:
- Board column changes: ~100-1000/day per project (bursty)
- Discussion comments: ~50-500/day per project
- Code review status changes: ~10-100/day per project
- Pipeline lock events: ~100-500/day per project

The existing Redis Streams + Elasticsearch infrastructure (documented in `03_implementation_plan.md`) can handle these volumes with margin.

### Detection Mechanism Scaling

Each adapter encapsulates its detection mechanism:

| Adapter | Mechanism | Scaling Characteristics |
|---------|-----------|------------------------|
| GitHub | Webhooks preferred, polling fallback | Webhook: no polling overhead; Polling: ~6000 req/hr limit |
| JIRA | Webhooks or JQL polling | Webhook preferred; Polling intervals configurable |
| Trello | Webhooks (required) | Rate limits: 300 req/10s, 100 req/10s per token |
| Azure DevOps | Service Hooks | Near real-time; queue-based delivery |

Adapters should implement **adaptive polling** when webhooks unavailable:
- Exponential backoff during low-activity periods
- Burst capacity during high-activity periods
- Respect per-vendor rate limits via existing `IRateLimiter`

### Horizontal Scaling Patterns

For multi-instance deployments:
1. **Event Deduplication**: Use `event_id` for idempotent processing
2. **Partition by Project**: Each instance handles subset of projects
3. **Leader Election**: For lock service coordination (Redis-based)

### Pattern 1: Event-Emitting Adapter (Decorator Pattern Extension)

```python
# src/codetoreum/ports/output/event_emitter.py
from abc import ABC, abstractmethod
from typing import Callable, List
from codetoreum.domain.events import CodetoreumEvent

class IEventEmitter(ABC):
    """Event emission capability for adapters."""
    
    @abstractmethod
    def on(self, event_type: str, handler: Callable[[CodetoreumEvent], None]) -> None:
        """Subscribe to events of a specific type."""
        pass
    
    @abstractmethod
    def off(self, event_type: str, handler: Callable[[CodetoreumEvent], None]) -> None:
        """Unsubscribe from events."""
        pass
    
    @abstractmethod
    def emit(self, event: CodetoreumEvent) -> None:
        """Emit an event to all subscribers."""
        pass
```

## ADR: Event-Based vs Polling-Based Orchestrator Integration

### Context

The orchestrator needs to react to external changes (work items moving between columns, new comments, PR approvals). Two approaches exist:
1. **Polling-based**: Orchestrator polls adapters periodically
2. **Event-based**: Adapters emit events, orchestrator subscribes

### Decision

**Adopt event-based adapters where detection mechanisms are adapter-internal.**

### Rationale

1. **Vendor Flexibility**: Each vendor has different optimal detection mechanisms (GitHub webhooks, JIRA JQL polling, Trello webhooks). Encapsulating detection inside adapters allows each to use its optimal approach.

2. **Clean Separation**: Orchestrator logic doesn't depend on detection method. A `workitem.column_changed` event is handled identically whether it came from a GitHub webhook or JIRA polling.

3. **Testability**: Mock adapters can emit events directly without simulating polling or webhook infrastructure. This enables deterministic, fast simulation tests.

4. **Consistency with Architecture**: The existing event-sourcing infrastructure (`IEventStore`, `EventBus`) already provides event distribution. Extending this to adapter events maintains architectural consistency.

### Consequences

- **Positive**: Cleaner orchestrator code, better testability, vendor flexibility
- **Negative**: Slightly more complex adapter implementations (must manage detection lifecycle)
- **Mitigation**: Provide base classes and mixins for common detection patterns

## ADR: Terminology Standardization

### Context

GitHub uses terms like "Issue", "Pull Request", "Projects v2 Status Field". JIRA uses "Issue", "Pull Request" (via Bitbucket), "Board". The codebase needs consistent terminology.

### Decision

**Adopt vendor-agnostic terminology as specified in the requirement:**

| Vendor Term | Standard Term |
|-------------|---------------|
| Issue (GitHub/JIRA) | Work Item |
| Issue Number | Work Item ID |
| Projects v2 | Project Board |
| Status Field | Column / Workflow State |
| Pull Request / Merge Request | Code Review |
| Issue Comment / PR Comment | Comment |

### Rationale

1. **Mental Model**: Developers working with Codetoreum shouldn't need to think in GitHub terms when the system supports multiple vendors.

2. **Interface Stability**: Using vendor-neutral terms in interfaces means interfaces don't change when adding vendors.

3. **Documentation Clarity**: Simulation scenarios and test cases use consistent terminology.

### Consequences

- **Positive**: Cleaner domain model, easier multi-vendor support
- **Negative**: Adapters must translate terminology at boundaries
- **Mitigation**: Use consistent mapping utilities; document term mappings per vendor

## Security Considerations

### Event Data Sensitivity

Events may contain sensitive information (work item titles, code review comments). Ensure:
1. Events in Redis Streams have appropriate TTL (2 hours per existing design)
2. Elasticsearch event indices follow retention policy (90 days)
3. Mock adapters don't log sensitive data in test output
4. Webhook endpoints validate signatures (GitHub: `X-Hub-Signature-256`)
